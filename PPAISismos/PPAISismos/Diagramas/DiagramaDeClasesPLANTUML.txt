' Paquete: Entidades
class Sesion <<entity>> {
    - fechaFin: DateTime
    - fechaInicio: DateTime
    usuarioLogueado: Usuario
    getUsuario(): Empleado
}

class Usuario <<entity>> {
    - nombreUsuario: string
    - contrasena: string
    empleado: Empleado
    getEmpleado(): Empleado
}

class Empleado <<entity>> {
    nombre
    apellido
    mail
    telefono
    rol
    
}

class Rol <<entity>>{
-nombre
-descripcionRol
+esResponsable()
}

class OrdenDeInspeccion <<entity>> {
    empleado: Empleado
      estacionSismologica: EstacionSismologica
      estadoOI: EstadoOI
      fechaHoraCierre: DateTime?
      - fechaHoraInicio: DateTime
      - fechaHoraFinalizacion: DateTime
      numeroOrden: int
      observacionCierre: string
      getNumeroOrden(): int
      getFechaFinalizacion(): DateTime
    
    esDeEmpleado(empleado: Empleado): bool
    verificarOIRealizada(): bool
    
    obtenerES(): EstacionSismologica
    getNombreEs(): string
   
    
    cerrarOI()
}

class EstadoOI <<entity>>{
      nombre: string
      esRealizada(): bool
      esCerrada(): bool
  }

class EstacionSismologica <<entity>>
{
- codigoEstacion: int
      - documentoCertificacionAdquirido: bool
      - latitud: double
      - longitud: double
      nombre: string
      - nroCertificacionAdquisicion: int
      getNombre(): string
}

class Sismografo <<entity>> {
     -fechaAdquisicion: DateTime
      identificadorSismografo: int
      - nroSerie: int
      - listaCambioEstadoSismografo: List<CambioEstadoSismografo>
      estacionSismologica: EstacionSismologica
      esTuES(es: EstacionSismologica): bool
      getIdentificador(): int

      ponerSismografoFueraServicio()
      buscarCambioEstadoActual()
}

class EstadoSismografo <<entity>> {
      - nombre: string
      + esFueraServicio(): bool
}

 class CambioEstadoSismografo <<entity>>{
      - fechaHoraFin: DateTime?
      - fechaHoraInicio: DateTime
      - estadoSismografo: EstadoSismografo
      motivoTipo: MotivoTipo
      
      esActual()
      setFechaFin()
      new()
      crearMotivosFueraServicio()
  }

class MotivoTipo <<entity>>{
   descripcion
   getDescripcion()
}

class MotivoFueraServicio <<entity>>{
   comentario
   tipoMotivo: MotivoTipo
   new()
}
